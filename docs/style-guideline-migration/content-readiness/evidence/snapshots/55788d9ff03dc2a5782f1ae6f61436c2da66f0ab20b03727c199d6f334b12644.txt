{%- comment -%}
  Monetta storefront layout (first delivery). Used by the redesigned templates (home, collection, search, product,
  cart) through the "layout" key of their JSON templates. It deliberately loads none of Atelier's base.css or
  scripts, so the approved design's CSS runs unchanged; the templates not yet ported keep layout/theme.liquid.
  Commerce stays Shopify's: catalog, cart and checkout come from the store; this file only frames them.

  Language: every text this theme writes is Spanish (Mexico), so the document declares es-MX instead of the store locale (the staging
  store's is English, which made a Spanish page announce itself as English). Fragments Shopify renders in the store's own language
  are marked individually with the monetta-store-lang snippet; the store's language setting is not touched and a multilingual
  store is out of scope.
{%- endcomment -%}
{%- liquid
  assign t = template.name
  assign has_page_bar = false
  assign dock_mode = 'nav'
  assign dock_active = ''
  if t == 'collection' or t == 'search' or t == 'cart'
    assign has_page_bar = true
  endif
  if t == 'product'
    assign dock_mode = 'purchase'
  elsif t == 'cart' and cart.item_count > 0
    assign dock_mode = 'checkout'
  endif
  case t
    when 'index'
      assign dock_active = 'home'
    when 'collection', 'search'
      assign dock_active = 'catalog'
    when 'cart'
      assign dock_active = 'cart'
  endcase
  assign og_image = page_image | default: shop.brand.logo
-%}

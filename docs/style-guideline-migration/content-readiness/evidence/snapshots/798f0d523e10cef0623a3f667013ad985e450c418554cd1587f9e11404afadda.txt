<section class="section container style-grid{% unless any_art %} style-grid--plain{% endunless %}" aria-labelledby="{{ heading_id }}" {{ section.shopify_attributes }}>

    id: heading_id,
    title: section.settings.heading,

    link_label: section.settings.link_label
  %}
  <div class="rail">
    <ul class="rail__track" role="list" aria-label="{{ section.settings.heading | escape }}">
      {%- for block in section.blocks -%}
        {%- liquid
          assign target = block.settings.collection
          assign href = block.settings.link | default: target.url
          assign label = block.settings.title | default: target.title

        -%}
        {%- if href != blank and label != blank -%}
          <li {{ block.shopify_attributes }}>
            <a class="style-tile{% unless art %} style-tile--all{% endunless %}" href="{{ href }}">

                {%- if art -%}
                  {% render 'monetta-image', image: art, widths: '240,360,480', sizes: '(min-width: 1024px) 25vw, 33vw', alt: '' %}

                  <span class="style-tile__circle">{% render 'monetta-icon', name: 'arrowForward', size: 24 %}</span>

              </span>

            </a>
          </li>
        {%- endif -%}
      {%- endfor -%}
    </ul>
  </div>
</section>

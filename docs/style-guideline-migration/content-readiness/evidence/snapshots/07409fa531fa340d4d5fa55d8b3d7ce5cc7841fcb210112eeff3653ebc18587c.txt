{%- doc -%}
  One bag line. Amounts, discounts and the quantity ceiling come from Shopify; nothing is recomputed here.
  The line is identified by its key (variant + properties + selling plan), never by position. The editable line also carries
  data-restore: the variant, quantity, properties (public and private) and selling plan it has right now, so that "Deshacer"
  can put back that very line and not just another purchase of the same variant.

  @param {object} item - A cart line item.
  @param {boolean} [compact] - Read-only variant for the "En tu bolsa" sheet.
{%- enddoc -%}
{%- liquid
  assign variant = item.variant
  assign max_qty = 99
  if variant.inventory_management == 'shopify' and variant.inventory_policy == 'deny'
    assign max_qty = variant.inventory_quantity
  endif
  assign discounted = false
  if item.original_price != item.final_price
    assign discounted = true
  endif
  assign marked_down = false
  if variant.compare_at_price > item.original_price
    assign marked_down = true
  endif
  assign restore_plan = 'null'
  if item.selling_plan_allocation
    assign restore_plan = item.selling_plan_allocation.selling_plan.id
  endif
-%}
{%- comment -%}`item.properties | json` serialises the pairs as an array, which would read back as "no properties"; write the object explicitly.{%- endcomment -%}
{%- capture restore_properties -%}{ {%- for property in item.properties -%}{{ property.first | json }}:{{ property.last | json }}{%- unless forloop.last -%},{%- endunless -%}{%- endfor -%} }{%- endcapture -%}
{%- capture restore_json -%}{"id":{{ item.variant_id }},"quantity":{{ item.quantity }},"properties":{{ restore_properties }},"selling_plan":{{ restore_plan }}}{%- endcapture -%}
<li class="cart-line"{% unless compact %} data-line="{{ item.key }}" data-restore="{{ restore_json | escape }}"{% endunless %} data-variant-id="{{ item.variant_id }}">

    {%- unless item.product.has_only_default_variant -%}
      <p class="cart-line__variant">{{ variant.title | escape }}</p>
    {%- endunless -%}
    {%- for property in item.properties -%}
      {%- unless property.last == blank or property.first contains '_' -%}
        <p class="cart-line__variant">{{ property.first | escape }}: {{ property.last | escape }}</p>
      {%- endunless -%}

        <span class="visually-hidden">, antes</span><s class="price-compare">{{ variant.compare_at_price | money_without_trailing_zeros }}</s>
      {%- endif -%}

    {%- else -%}
      <form class="cart-line__controls" action="{{ routes.cart_change_url }}" method="post" data-line-form>
        <input type="hidden" name="id" value="{{ item.key }}">

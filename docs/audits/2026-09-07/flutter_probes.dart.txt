void main() {
 test('AUDIT full sync does not remove absent products', () async {
  final db=openTestDatabase();
  addTearDown(db.close);
  final transport=fixture.FakeTransport([fixture.page([fixture.productNode(id:'1',updatedAt:'2026-08-01T00:00:00Z')]),fixture.page([])]);
  final repo=CatalogRepository(cache:CatalogCacheStore(db),cursorStore:SyncCursorStore(db),rules:NormalizationRules.fromJson({}),catalogSync:CatalogSync(client:StorefrontClient(config:const StorefrontConfig(shopDomain:'example.myshopify.com',accessToken:'fake'),transport:transport.call),fullResyncInterval:Duration.zero));

 });
 test('AUDIT truncated pagination is labelled successful full sync', () async {
  final transport=fixture.FakeTransport([fixture.page([fixture.productNode(id:'1',updatedAt:'2026-08-01T00:00:00Z')],hasNextPage:true,endCursor:'still-more')]);

  expect(result.wasFullSync,true);
  expect(result.cursor.lastFullSyncAt,isNotNull);
 });
 test('AUDIT bag and wishlist readers include another account rows', () async {
  final db=openTestDatabase(); addTearDown(db.close);
  final bag=BagStore(db); final wishes=WishlistStore(db);
  await bag.upsert(productId:'product-A',variantId:'variant-A',quantity:1,accountId:'customer-A');
  await wishes.add('product-A',accountId:'customer-A');
  expect((await bag.all()).keys,contains('product-A'));
  expect(await wishes.allProductIds(),contains('product-A'));
  // Neither reader accepts the current account. Both return A to any caller.
 });
